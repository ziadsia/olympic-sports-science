function sanitizeYouTubeId(rawId) { if (!rawId || typeof rawId !== "string") return null; const cleaned = rawId.trim().split(/[?&#/]/)[0]; return /^[a-zA-Z0-9_-]{6,20}$/.test(cleaned) ? cleaned : null; } function extractYouTubeId(urlObj) { const host = urlObj.hostname.toLowerCase(); if (host === "youtu.be" || host.endsWith(".youtu.be")) { const firstSeg = urlObj.pathname.replace(/^\/+/, "").split("/")[0]; return sanitizeYouTubeId(firstSeg); } if ( host === "youtube.com" || host.endsWith(".youtube.com") || host === "youtube-nocookie.com" || host.endsWith(".youtube-nocookie.com") ) { const vParam = urlObj.searchParams.get("v"); if (vParam) { const idFromV = sanitizeYouTubeId(vParam); if (idFromV) return idFromV; } const parts = urlObj.pathname.split("/").filter(Boolean); const specialIdx = parts.findIndex((p) => ["embed", "shorts", "live", "v", "e"].includes(p.toLowerCase()) ); if (specialIdx !== -1 && parts[specialIdx + 1]) { return sanitizeYouTubeId(parts[specialIdx + 1]); } } return null; } function isYouTubeHost(hostname) { const h = (hostname || "").toLowerCase(); return ( h === "youtube.com" || h.endsWith(".youtube.com") || h === "youtu.be" || h.endsWith(".youtu.be") || h === "youtube-nocookie.com" || h.endsWith(".youtube-nocookie.com") ); } function normalizeVideoUrl(rawUrl) { if (!rawUrl || !rawUrl.trim()) { return { ok: false, reason: "لا يوجد رابط فيديو مضاف لهذه المحاضرة حالياً." }; } let parsed; try { parsed = new URL(rawUrl.trim()); } catch (e) { return { ok: false, reason: "رابط الفيديو غير صالح." }; } if (parsed.protocol !== "https:" && parsed.protocol !== "http:") { return { ok: false, reason: "صيغة رابط الفيديو غير مدعومة." }; } if (isYouTubeHost(parsed.hostname)) { const ytId = extractYouTubeId(parsed); if (!ytId) { return { ok: false, reason: "تعذر استخراج معرّف فيديو YouTube من الرابط المدخل. تأكد من نسخ رابط الفيديو الصحيح." }; } return { ok: true, safeUrl: `https://www.youtube.com/embed/${ytId}?rel=0&modestbranding=1` }; } return { ok: true, safeUrl: parsed.toString() }; }
